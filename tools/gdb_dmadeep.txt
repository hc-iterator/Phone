# tools\gdb_dmadeep.txt —— DVI 引擎卡死的深度现场（第 2 版）
#
# 第 1 版教训：GDB 批处理模式下，脚本里任何一条命令报错，
#   【后面的命令全部不再执行】。所以：
#     1. 不许用 printf --（不是合法语法）
#     2. 不许在双引号里再嵌双引号
#     3. 最重要的段落放最前面
#
# dma_debug_hw / dma_hw 都是宏不是符号 ⇒ 用硬地址：
#   DMA_BASE = 0x50000000，通道 n = base + n*0x40
#   调试区 = 0x50000800 + n*0x40（+0 dbg_ctdreq，+4 dbg_tcr）

set confirm off
set pagination off
set height 0
target extended-remote localhost:3333
monitor halt

printf "\n===== 1) 两个核心的位置（最重要）=====\n"
info threads
thread apply all bt
thread apply all info registers control primask basepri faultmask msp psp pc xpsr

printf "\n===== 2) DVI 实例状态 =====\n"
printf "dvi_started=%d  frame_count=%u  v_ctr=%u  v_state=%u  late_scanline_ctr=%u\n", dvi0.dvi_started, dvi0.dvi_frame_count, dvi0.timing_state.v_ctr, dvi0.timing_state.v_state, dvi0.late_scanline_ctr
printf "data_island_is_enabled=%d  scanline_is_enabled=%d  audio_freq=%d\n", dvi0.data_island_is_enabled, dvi0.scanline_is_enabled, dvi0.audio_freq
printf "h_active=%u 期望 dbg_tcr=%u\n", dvi0.timing->h_active_pixels, dvi0.timing->h_active_pixels / 2

printf "\n===== 3) dma_cb_t 与 dvi_scanline_dma_list 的真实尺寸 =====\n"
p sizeof(dma_cb_t)
p sizeof(struct dvi_scanline_dma_list)
p DVI_SYNC_LANE_CHUNKS_WITH_AUDIO
p DVI_NOSYNC_LANE_CHUNKS_WITH_AUDIO
p &dvi0.dma_list_active
p &dvi0.dma_list_active.l0
p &dvi0.dma_list_active.l1
p &dvi0.dma_list_active.l2
p &dvi0.dma_list_vblank_nosync
p &dvi0.dma_list_active_blank

printf "\n===== 4) dma_list_active 原始内存（每行 8 字）=====\n"
x/72xw &dvi0.dma_list_active

printf "\n===== 5) 全部 16 个 DMA 通道 =====\n"
set $i = 0
while $i < 16
  printf "ch%02d CTRL=%08x TRANS=%08x READ=%08x WRITE=%08x | dbg_tcr=%08x dbg_ctdreq=%08x\n", $i, *(unsigned int*)(0x50000000+$i*0x40+0x0c), *(unsigned int*)(0x50000000+$i*0x40+0x08), *(unsigned int*)(0x50000000+$i*0x40+0x00), *(unsigned int*)(0x50000000+$i*0x40+0x04), *(unsigned int*)(0x50000800+$i*0x40+0x04), *(unsigned int*)(0x50000800+$i*0x40+0x00)
  set $i = $i + 1
end

printf "\n===== 6) DMA 中断与 PIO =====\n"
printf "INTR=%08x INTE0=%08x INTS0=%08x INTE1=%08x INTS1=%08x\n", *(unsigned int*)0x50000400, *(unsigned int*)0x50000404, *(unsigned int*)0x5000040c, *(unsigned int*)0x50000414, *(unsigned int*)0x5000041c
printf "PIO0 CTRL=%08x FDEBUG=%08x\n", *(unsigned int*)0x50200000, *(unsigned int*)0x50200008
printf "PIO1 CTRL=%08x FDEBUG=%08x\n", *(unsigned int*)0x50300000, *(unsigned int*)0x50300008
printf "PIO2 CTRL=%08x FDEBUG=%08x\n", *(unsigned int*)0x50400000, *(unsigned int*)0x50400008

printf "\n===== 7) 存活信标与故障寄存器 =====\n"
printf "g_dvi_loop_frames=%u  g_probe_marks=%u  g_probe_ctrl4=%08x\n", g_dvi_loop_frames, g_probe_marks, g_probe_ctrl[4]
printf "CFSR=%08x HFSR=%08x MMFAR=%08x BFAR=%08x SHCSR=%08x MPU_CTRL=%08x\n", *(unsigned int*)0xE000ED28, *(unsigned int*)0xE000ED2C, *(unsigned int*)0xE000ED34, *(unsigned int*)0xE000ED38, *(unsigned int*)0xE000ED24, *(unsigned int*)0xE000ED94

monitor resume
detach
quit
