# tools\gdb_panic.txt —— 抓 panic 的调用栈
#
# 现场：板子 halt 在 __breakpoint()（pico/platform.h:119），
#       也就是 SDK 的 panic() 路径。要看【谁】调的 panic。
#
# 纪律：不 p 宏、不用 printf --、不嵌双引号、不调目标函数、要看的放最前。

set confirm off
set pagination off
set height 0
target extended-remote localhost:3333
monitor halt

printf "\n===== 1) panic 从哪儿来 =====\n"
bt

printf "\n===== 2) 断在哪 =====\n"
info threads

printf "\n===== 3) 实验状态 =====\n"
printf "wt_done=%u resumed_from=%d hung_mask=0x%04x end_cfg_ok=%u\n", g_wt_done, g_wt_resumed_from, g_wt_hung_mask, g_wt_end_config_ok
printf "diag: done=%u sdk_cs=GPIO%u timing=0x%08x clkdiv=%u rxdelay=%u maxsel=%u mindesel=%u\n", g_psram_diag_done, g_psram_diag_sdk_cs_gpio, g_psram_diag_qmi_timing, g_psram_diag_divisor, g_psram_diag_rxdelay, g_psram_diag_max_select, g_psram_diag_min_deselect
printf "marks=%u\n", g_probe_marks
printf "SRAM池 free=%u used=%u\n", g_pool_sram.free_bytes, g_pool_sram.used_bytes
printf "kres 自检 status=%u steps=%u fail_at=%u\n", g_kres_selftest_status, g_kres_selftest_steps, g_kres_selftest_fail_at

monitor resume
detach
quit
